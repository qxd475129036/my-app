# Environment Theme System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add 4 environment-specific gradient themes (local/green, dev/yellow, stg/purple, prd/light-blue) switched via `?env=` URL parameter.

**Architecture:** `LayoutClient` reads `?env=` from URL search params and sets `data-env` on `<html>`. CSS `[data-env]` selectors override `--accent`, `--accent-light`, `--accent-hover`, `--sidebar-bg`, `--sidebar-active`, and add `--env-gradient`. Sidebar uses these CSS variables instead of hardcoded hex colors. Navbar already uses `--accent` via Tailwind classes, so it inherits changes automatically.

**Tech Stack:** Next.js 16 (App Router), TypeScript, Tailwind CSS 4, CSS custom properties

### Task 2: Add environment detection in LayoutClient

**Files:**
- Modify: `src/app/components/LayoutClient.tsx`

- [ ] **Step 1: Add `useSearchParams` and `useEffect` to read `?env=` and set `data-env`**

Replace `src/app/components/LayoutClient.tsx` with:

```tsx
"use client";

import { usePathname, useSearchParams } from "next/navigation";
import { useEffect, useState } from "react";
import { Sidebar } from "./Sidebar";
import { Footer } from "./Footer";

const VALID_ENVS = ["local", "dev", "stg", "prd"] as const;
type Env = (typeof VALID_ENVS)[number];

function resolveEnv(raw: string | null): Env {
  if (raw && VALID_ENVS.includes(raw as Env)) {
    return raw as Env;
  }
  return "local";
}

export function LayoutClient({ children }: { children: React.ReactNode }) {
  const pathname = usePathname();
  const searchParams = useSearchParams();
  const isLoginPage = pathname.startsWith("/login");
  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);

  useEffect(() => {
    const env = resolveEnv(searchParams.get("env"));
    document.documentElement.dataset.env = env;
  }, [searchParams]);

  if (isLoginPage) {
    return <>{children}</>;
  }

  return (
    <div className="min-h-screen flex">
      <Sidebar
        collapsed={sidebarCollapsed}
        onToggle={() => setSidebarCollapsed((prev) => !prev)}
      />
      <div
        className={`flex flex-col flex-1 min-w-0 transition-all duration-300 ${
          sidebarCollapsed ? "ml-16" : "ml-64"
        }`}
      >
        <main className="flex-1 bg-gray-50">{children}</main>
        <Footer />
      </div>
    </div>
  );
}
```

import { Suspense } from "react";
import { Geist, Geist_Mono } from "next/font/google";

const geistSans = Geist({

  subsets: ["latin"],
  display: "swap",
});

const geistMono = Geist_Mono({
  variable: "--font-geist-mono",
  subsets: ["latin"],
  display: "swap",
});

export const metadata: Metadata = {
  title: "MyApp",
  description: "Next.js App with Authentication",
};

export default function RootLayout({
  children,
}: Readonly<{
  children: React.ReactNode;
}>) {
  return (
    <html
      lang="en"
      className={`${geistSans.variable} ${geistMono.variable} h-full antialiased`}
    >
      <body className="min-h-full flex flex-col font-sans">
        <AppSessionProvider>
          <Suspense fallback={null}>
            <LayoutClient>{children}</LayoutClient>
          </Suspense>
        </AppSessionProvider>
      </body>
    </html>
  );
}
